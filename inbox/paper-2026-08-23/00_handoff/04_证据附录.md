## §1 探针脚本（可直接复跑）

### 1.1 站点 agent 配置探针

下发方式：本地写脚本 → `scp` → `ssh <host> "bash /tmp/_probe.sh"`（守手册 R14 铁律：PowerShell→ssh 多层转义必崩）

```bash
#!/bin/bash
set -uo pipefail
echo "##### HOST=$(hostname) #####"

echo "--- [1] opencode config files ---"
ls -la "$HOME/.config/opencode/" 2>/dev/null

echo "--- [2] provider keys ---"
grep -nE '^ {4}"[A-Za-z0-9_-]+"[[:space:]]*:[[:space:]]*\{' "$HOME/.config/opencode/opencode.jsonc" 2>/dev/null

echo "--- [3] model / limit / baseURL ---"
grep -nE '"model"|"context"|"output"|baseURL' "$HOME/.config/opencode/opencode.jsonc" 2>/dev/null

echo "--- [4] plugins ---"
sed -n '/"plugin"/,/]/p' "$HOME/.config/opencode/opencode.jsonc" 2>/dev/null
echo "--- [4b] skills ---"
ls "$HOME/.claude/skills/" 2>/dev/null | tr '\n' ' '; echo
echo "--- [4c] ARS reachability ---"
for d in skills commands agents plugins; do
  t="$HOME/.config/opencode/$d"
  if [ -d "$t" ]; then
    tot=$(find "$t" -maxdepth 1 -type l | wc -l)
    brk=$(find "$t" -maxdepth 1 -type l ! -exec test -e {} \; -print 2>/dev/null | wc -l)
    echo "ars.$d links=$tot broken=$brk"
  fi
done

echo "--- [5] claude settings env/model ---"
grep -nE 'MAX_CONTEXT_TOKENS|ANTHROPIC_BASE_URL|"model"|DISABLE_TOOLS|DISABLE_AUTOUPDATER' "$HOME/.claude/settings.json" 2>/dev/null
echo "--- [5b] modelOverrides ---"
sed -n '/modelOverrides/,/}/p' "$HOME/.claude/settings.json" 2>/dev/null

echo "--- [6] llama-instances conf ---"
for f in /etc/llama-instances/*.env; do
  [ -f "$f" ] || continue
  printf '%s : ' "$(basename "$f")"
  grep -E '^(CTX|PORT|BACKEND|RPC_TARGET)=' "$f" 2>/dev/null | tr '\n' ' '
  echo
done

echo "--- [7] engine / versions ---"
ss -tlnp 2>/dev/null | grep -c llama-server || true
opencode --version 2>/dev/null | head -1
claude --version 2>/dev/null | head -1

echo "--- [8] workspaces ---"
ls -la "$HOME/agent-workspaces/" 2>/dev/null
du -sh "$HOME/agent-workspaces/paper/" 2>/dev/null
```
